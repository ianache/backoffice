**bff/src/routes/tenants.ts** — implement the proxy router exactly per RESEARCH.md Pattern 3:
```typescript
import { Router } from 'express'
import { createProxyMiddleware } from 'http-proxy-middleware'
import { requireAuth } from '../middleware/auth.js'
import { requireRole } from '../middleware/roles.js'
import { config } from '../config/index.js'

export const tenantsRouter = Router()

tenantsRouter.use(
  requireAuth,
  requireRole('PlatformAdmin'),
  createProxyMiddleware({
    target: config.backendUrl,
    changeOrigin: true,
    on: {
      proxyReq: (proxyReq, req) => {
        proxyReq.setHeader('X-Internal-Secret', config.internalSecret)
        proxyReq.setHeader('X-User-Sub', (req as any).user?.sub ?? '')
        proxyReq.setHeader('X-User-Roles', ((req as any).user?.roles ?? []).join(','))
      },
    },
  })
)
```

IMPORTANT path note (RESEARCH.md Pitfall 4): When BFF mounts this router at `/tenants` and the proxy forwards to the backend, the `/tenants` prefix is preserved in the forwarded URL. The Python backend router also uses `prefix="/tenants"`, so paths match correctly: BFF receives `/tenants/` → proxy forwards to `http://localhost:8000/tenants/` → Python router handles `/tenants/`. Do NOT add `pathRewrite` unless this breaks.
