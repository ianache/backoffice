export async function addSegmentToFlag(flagId: number, segmentId: number): Promise<Segment> {
  const { data } = await api.post(`/flags/${flagId}/segments`, { segment_id: segmentId })
  return data
}

export async function getSegmentsByFlag(flagId: number): Promise<Segment[]> {
  const { data } = await api.get(`/flags/${flagId}/segments`)
  return data
}

1. Read `FlagDrawer.vue` first to understand its current structure (it has a `open` prop and calls `formRef.value?.handleSave()`).

2. Import store and service at top:
   ```typescript
   import { useFeatureFlagsStore } from '../../stores/flags'
   import { getSegmentsByFlag, addSegmentToFlag } from '../../services/flags'
   ```

3. Add reactive state for linked segments and a fetch on drawer open:
   ```typescript
   const flagsStore = useFeatureFlagsStore()
   const linkedSegmentIds = ref<number[]>([])
   
   // Fetch segments list (for SegmentPicker dropdown) + linked segments for edit mode
   watch(
     () => props.open,
     async (isOpen) => {
       if (isOpen) {
         await flagsStore.fetchSegments()
         if (props.flag?.id) {
           const linked = await getSegmentsByFlag(props.flag.id)
           linkedSegmentIds.value = linked.map(s => s.id)
         } else {
           linkedSegmentIds.value = []
         }
       }
     }
   )
   ```

4. Pass new props to FlagForm in template:
   ```html
   <FlagForm
     ref="formRef"
     :flag="props.flag"
     :segments="flagsStore.segments"
     :linked-segment-ids="linkedSegmentIds"
     @save="handleSave"
     @cancel="emit('close')"
   />
   ```

5. In the `handleSave` handler (after the existing `flagsStore.createFlag()` or `updateFlag()` call), attach segments. The handler receives `payload: FlagPayload` from FlagForm's emit. After saving, read `formRef.value?.selectedSegmentIds` and call `addSegmentToFlag` for each new ID:
   ```typescript
   async function handleSave(payload: FlagPayload) {
     try {
       let savedFlag: FeatureFlag
       if (props.flag) {
         savedFlag = await flagsStore.updateFlag(props.flag.id, payload)
       } else {
         savedFlag = await flagsStore.createFlag(payload)
       }
       // Link selected segments to the saved flag (FLAG-06)
       const selectedIds = formRef.value?.selectedSegmentIds ?? []
       for (const segmentId of selectedIds) {
         await addSegmentToFlag(savedFlag.id, segmentId)
       }
       emit('saved', savedFlag)
       emit('close')
     } catch (err: any) {
       // surface error via toast if available, or console.error
       console.error('Save flag failed:', err)
     }
   }
   ```

Note: `addSegmentToFlag` is idempotent on the backend (returns 201 or existing link). For Phase 4, only addition is needed — segment removal from a flag is deferred to Phase 5 (Rule Builder).
