```typescript
anyOf: (actual, expected) => {
  if (Array.isArray(actual)) {
    const expectedArr = Array.isArray(expected) ? expected : []
    return actual.some((v) => expectedArr.includes(v))
  }
  return Array.isArray(expected) && expected.includes(actual)
},
```
